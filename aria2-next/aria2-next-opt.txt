Usage: aria2-next [OPTIONS] [URI | MAGNET | TORRENT | METALINK | ED2K]...

 -d, --dir=path
        Directory to store downloaded files.
        Default: .
 -o, --out=path
        File name of the downloaded file, relative to --dir.
 -i, --input-file=path
        Read URIs and per-task options from FILE ('-' for stdin).
 -j, --max-concurrent-downloads=value
        Maximum number of parallel downloads.
        Default: 5
 -c, --continue
        Continue a partially downloaded file that this engine did not start.
        Default: false
 -V, --check-integrity
        Verify existing data with piece hashes or --checksum.
        Default: false
 -l, --log=path
        Write the log to FILE ('-' for stdout).
 -q, --quiet
        Do not print to the console.
        Default: false
 -D, --daemon
        Run in the background. Leave it off under a service manager (systemd, NSSM, WinSW), which tracks the process it started.
        Default: false
     --conf-path=path
        Configuration file (default: $XDG_CONFIG_HOME/aria2/aria2.conf).
     --no-conf
        Do not load a configuration file.
        Default: false
 -v, --version
        Print the version and exit.
        Default: false
 -h, --help=value
        Print help for a tag (#basic, #http, #bittorrent, #ed2k, #media, #rpc, #advanced, #all) or a keyword.
        Default: #basic
 -s, --split=value
        Connections used for one download.
        Default: 16
 -x, --max-connection-per-server=value
        Connections to one host for one download.
        Default: 16
 -k, --min-split-size=size
        Smallest range that is split for another connection.
        Default: 1M
 -t, --timeout=value
        Seconds without received data before a connection is retried.
        Default: 60
     --connect-timeout=value
        Seconds allowed for connecting.
        Default: 60
 -m, --max-tries=value
        Consecutive failures without progress before the task fails (0: unlimited).
        Default: 5
     --retry-wait=value
        Seconds to wait before retrying (0: automatic back-off).
        Default: 0
     --lowest-speed-limit=size
        Retry a connection slower than this many bytes/s (0: off).
        Default: 0
     --max-download-limit=size
        Download speed limit per task in bytes/s (0: unlimited).
        Default: 0
     --max-overall-download-limit=size
        Total download speed limit in bytes/s (0: unlimited).
        Default: 0
 -U, --user-agent=value
        User-Agent request header.
        Default: aria2-next
     --referer=value
        Referer request header ('*' uses the download URI).
     --header=value
        Additional request header; repeatable.
     --http-user=value
        HTTP user name.
     --http-passwd=value

     --http-no-cache

     --http-accept-gzip

     --save-cookies=path
        Save cookies to FILE in Netscape format.
     --check-certificate
        Verify TLS certificates.

     --ca-certificate=path

     --certificate=path

     --private-key=path

     --min-tls-version=TLSv1.1|TLSv1.2|TLSv1.3
        Minimum TLS version.

     --all-proxy=value
        Proxy for all protocols: [http://][USER:PASS@]HOST[:PORT], socks5://...
     --all-proxy-user=value
        User for --all-proxy.
     --all-proxy-passwd=value
        Password for --all-proxy.
     --http-proxy=value

     --http-proxy-user=value
        User for --http-proxy.
     --http-proxy-passwd=value
        Password for --http-proxy.
     --https-proxy=value
        Proxy for HTTPS.
     --https-proxy-user=value
        User for --https-proxy.
     --https-proxy-passwd=value
        Password for --https-proxy.
     --no-proxy=value

     --dns-over-https=value

     --interface=value
        Bind connections to an interface, IP address or host name.
     --disable-ipv6
        Do not use IPv6.

     --netrc-path=path

 -n, --no-netrc
        Ignore netrc.
        Default: false
     --sftp-user=value
        SFTP user name.
     --sftp-passwd=value
        SFTP password.
     --ssh-host-key-sha256=value
        Expected base64 SHA-256 fingerprint of the SFTP host key.
 -R, --remote-time
        Apply the server's Last-Modified time to the file.
        Default: false
     --checksum=value
        Expected checksum TYPE=DIGEST (sha-1, sha-224, sha-256, sha-384, sha-512, md5).
     --max-file-not-found=value

        Name suggested by the caller.

        Origin of --filename-hint; browser names win over Content-Disposition.

        Check availability without downloading.

     --allow-overwrite
        Restart from scratch when the output file exists.
        Default: false
     --auto-file-renaming
        Use a numbered name when the output file exists.
        Default: true
     --incomplete-suffix=value
        Suffix of unfinished HTTP, FTP and SFTP files, removed when they complete (for example .part).
     --auto-file-renaming-style=dot|parentheses
        Numbered names: dot (name.1.ext, as aria2) or parentheses (name (1).ext, as browsers).

        Default: none
     --remove-control-file
        Ignore stored progress and start over.

     --force-save
        Save the task in the session even when it is finished.
        Default: false
     --save-not-found
        Save tasks that failed with 404 in the session.
